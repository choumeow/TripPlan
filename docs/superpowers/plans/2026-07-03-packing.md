# Packing (#4) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the trip Packing tab — a collaborative checklist split into a shared group list and a per-person personal list (real members' lists private to them; ghost joiners' lists visible to all contributors), organised by category with a packed checkbox.

**Architecture:** One `packing_items` table with a `scope` discriminator (`shared`|`personal`) and an `owner_member_id`. Per-user personal-visibility is enforced by RLS using two `SECURITY DEFINER` helpers (`member_is_ghost`, `member_user_id`). "Anyone can tick a shared item packed" is a `SECURITY DEFINER` RPC (`set_packing_packed`) — the one carve-out from the stricter host/editor write RLS. The React page is a sidebar (master-detail) layout: a rail of lists + a right pane showing the selected list, following the existing slice patterns (pure `lib/`, react-query hooks invalidating `['trip', tripId]`, CSS Modules, TDD).

**Tech Stack:** React 19 + Vite, Supabase (Postgres + RLS + RPC), `@tanstack/react-query`, `react-router-dom`, CSS Modules, Vitest + React Testing Library.

**Spec:** `docs/superpowers/specs/2026-07-03-packing-design.md`

```sql
create table if not exists public.packing_items (
  id                 uuid primary key default gen_random_uuid(),
  trip_id            uuid not null references public.trips(id) on delete cascade,
  scope              text not null check (scope in ('shared','personal')),
  owner_member_id    uuid references public.trip_members(id) on delete cascade,
  category           text not null check (category in
                       ('documents','clothing','toiletries','electronics','gear','other')),
  title              text not null,
  assignee_member_id uuid references public.trip_members(id) on delete set null,
  packed             boolean not null default false,
  sort_order         int not null default 0,
  created_by         uuid references public.trip_members(id) on delete set null,
  created_at         timestamptz not null default now()
);
create index if not exists packing_items_trip_id_idx on public.packing_items (trip_id);
alter table public.packing_items enable row level security;
grant select, insert, update, delete on public.packing_items to authenticated;

-- ── helpers (SECURITY DEFINER → bypass trip_members RLS, no recursion) ────
create or replace function public.member_is_ghost(member_id uuid)
returns boolean language sql security definer stable set search_path = public as $$
  select user_id is null from public.trip_members where id = member_id
$$;
create or replace function public.member_user_id(member_id uuid)
returns uuid language sql security definer stable set search_path = public as $$
  select user_id from public.trip_members where id = member_id
$$;
grant execute on function public.member_is_ghost(uuid) to authenticated;
grant execute on function public.member_user_id(uuid) to authenticated;

-- ── RLS: SELECT (per-user personal visibility) ───────────────────────────
create policy "packing_select" on public.packing_items for select using (
  public.is_trip_member(trip_id) and (
    scope = 'shared'
    or public.member_user_id(owner_member_id) = auth.uid()
    or public.member_is_ghost(owner_member_id)
  )
);

-- ── RLS: full write (add / edit / delete) ────────────────────────────────
-- shared → host/editor; ghost personal → host/editor; own personal → owner (any role)
create policy "packing_insert" on public.packing_items for insert with check (
  public.is_trip_member(trip_id) and (
       (scope = 'shared'   and public.trip_role(trip_id) in ('host','editor'))
    or (scope = 'personal' and public.member_is_ghost(owner_member_id)
                          and public.trip_role(trip_id) in ('host','editor'))
    or (scope = 'personal' and public.member_user_id(owner_member_id) = auth.uid())
  )
);
create policy "packing_update" on public.packing_items for update
  using (
    public.is_trip_member(trip_id) and (
         (scope = 'shared'   and public.trip_role(trip_id) in ('host','editor'))
      or (scope = 'personal' and public.member_is_ghost(owner_member_id)
                            and public.trip_role(trip_id) in ('host','editor'))
      or (scope = 'personal' and public.member_user_id(owner_member_id) = auth.uid())
    )
  )
  with check (
    public.is_trip_member(trip_id) and (
         (scope = 'shared'   and public.trip_role(trip_id) in ('host','editor'))
      or (scope = 'personal' and public.member_is_ghost(owner_member_id)
                            and public.trip_role(trip_id) in ('host','editor'))
      or (scope = 'personal' and public.member_user_id(owner_member_id) = auth.uid())
    )
  );
create policy "packing_delete" on public.packing_items for delete
  using (
    public.is_trip_member(trip_id) and (
         (scope = 'shared'   and public.trip_role(trip_id) in ('host','editor'))
      or (scope = 'personal' and public.member_is_ghost(owner_member_id)
                            and public.trip_role(trip_id) in ('host','editor'))
      or (scope = 'personal' and public.member_user_id(owner_member_id) = auth.uid())
    )
  );

-- ── RPC: "if you can see it, you can tick it" (packed only) ───────────────
create or replace function public.set_packing_packed(p_item_id uuid, p_packed boolean)
returns void language plpgsql security definer set search_path = public as $$
declare v_trip uuid; v_scope text; v_owner uuid;
begin
  select trip_id, scope, owner_member_id into v_trip, v_scope, v_owner
    from public.packing_items where id = p_item_id;
  if v_trip is null then raise exception 'not found'; end if;
  if not (public.is_trip_member(v_trip) and (
        v_scope = 'shared'
     or public.member_user_id(v_owner) = auth.uid()
     or public.member_is_ghost(v_owner)
  )) then raise exception 'forbidden'; end if;
  update public.packing_items set packed = p_packed where id = p_item_id;
end $$;
grant execute on function public.set_packing_packed(uuid, boolean) to authenticated;
```

```js
import { PACKING_CATEGORY_ORDER } from './packingCategories'

// Split all visible packing rows into the lists a given user works with.
// RLS already strips other real members' personal rows; we filter defensively too.
export function partition(items, members, authUserId) {
  const list = items ?? []
  const mems = members ?? []
  const myMemberId = mems.find((m) => m.user_id === authUserId)?.id ?? null
  const shared = list.filter((i) => i.scope === 'shared')
  const mine = list.filter((i) => i.scope === 'personal' && i.owner_member_id === myMemberId)
  const ghostLists = mems
    .filter((m) => m.user_id == null)
    .map((m) => ({
      member: m,
      items: list.filter((i) => i.scope === 'personal' && i.owner_member_id === m.id),
    }))
  return { shared, mine, myMemberId, ghostLists }
}

// Group items under category keys in fixed order; drop empty categories.
export function groupByCategory(items) {
  const list = items ?? []
  return PACKING_CATEGORY_ORDER
    .map((key) => ({ key, items: list.filter((i) => i.category === key) }))
    .filter((g) => g.items.length > 0)
}

// {packed, total} across the given items.
export function progress(items) {
  const list = items ?? []
  return { packed: list.filter((i) => i.packed).length, total: list.length }
}
```

```js
import { useMutation, useQueryClient } from '@tanstack/react-query'
import { supabase } from '../lib/supabaseClient'

export function useUpsertPackingItem(tripId) {
  const queryClient = useQueryClient()
  return useMutation({
    mutationFn: async (item) => {
      const { id, createdBy, ...fields } = item
      const query = id
        ? supabase.from('packing_items').update(fields).eq('id', id)
        : supabase.from('packing_items').insert({ ...fields, trip_id: tripId, created_by: createdBy ?? null })
      const { data, error } = await query.select().single()
      if (error) throw error
      return data
    },
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['trip', tripId] }),
  })
}
```

```js
import { useMutation, useQueryClient } from '@tanstack/react-query'
import { supabase } from '../lib/supabaseClient'

export function useTogglePacked(tripId) {
  const queryClient = useQueryClient()
  const key = ['trip', tripId]
  return useMutation({
    mutationFn: async ({ id, packed }) => {
      const { error } = await supabase.rpc('set_packing_packed', { p_item_id: id, p_packed: packed })
      if (error) throw error
    },
    onMutate: async ({ id, packed }) => {
      await queryClient.cancelQueries({ queryKey: key })
      const previous = queryClient.getQueryData(key)
      if (previous) {
        queryClient.setQueryData(key, {
          ...previous,
          packing_items: (previous.packing_items ?? []).map((r) => (r.id === id ? { ...r, packed } : r)),
        })
      }
      return { previous }
    },
    onError: (_e, _v, ctx) => { if (ctx?.previous) queryClient.setQueryData(key, ctx.previous) },
    onSettled: () => queryClient.invalidateQueries({ queryKey: key }),
  })
}
```

```js
import styles from './PackingRail.module.css'

function pct({ packed, total }) {
  return total ? Math.round((packed / total) * 100) : 0
}

function ListButton({ entry, selected, onSelect }) {
  return (
    <button type="button"
      className={`${styles.litem} ${selected === entry.id ? styles.on : ''}`}
      aria-current={selected === entry.id}
      onClick={() => onSelect(entry.id)}>
      <span className={styles.ic} aria-hidden="true">{entry.glyph}</span>
      <span className={styles.body}>
        <span className={styles.nm}>{entry.label}</span>
        <span className={styles.mini}><span style={{ width: `${pct(entry)}%` }} /></span>
      </span>
      {entry.tag
        ? <span className={`${styles.tag} ${entry.ghost ? styles.ghostTag : styles.privTag}`}>{entry.tag}</span>
        : <span className={styles.mt}>{entry.packed}/{entry.total}</span>}
    </button>
  )
}

            {ghosts.map((e) => <ListButton key={e.id} entry={e} selected={selected} onSelect={onSelect} />)}
```

```js
import { useState } from 'react'
import { useOutletContext } from 'react-router-dom'
import { useAuth } from '../auth/AuthContext'
import { callerMember, canWrite } from '../lib/tripAccess'
import { memberName, initials } from '../lib/display'
import { partition, progress } from '../lib/packing'
import { PackingRail } from '../components/PackingRail'
import { PackingList } from '../components/PackingList'
import { PackingItemEditorModal } from '../components/PackingItemEditorModal'
import { useTogglePacked } from '../hooks/useTogglePacked'
import { useDeletePackingItem } from '../hooks/useDeletePackingItem'
import styles from './Packing.module.css'

const HINTS = {
  shared: 'Visible to everyone. Anyone can tick an item packed; host & editors add, assign or delete.',
  personal: 'Private to you — nobody else can see this list.',
  ghost: 'Ghost traveller (no login) — visible to all contributors, managed by host & editors.',
}

export function Packing() {
  const { trip } = useOutletContext()
  const { user } = useAuth()
  const me = callerMember(trip, user?.id)
  const isWriter = canWrite(me?.role)
  const part = partition(trip.packing_items, trip.trip_members, user?.id)

  const [selected, setSelected] = useState('shared')
  const [editor, setEditor] = useState(null)
  const toggle = useTogglePacked(trip.id)
  const del = useDeletePackingItem(trip.id)

  // Rail entries (shared + your list + one per ghost).
  const entries = [
    { id: 'shared', label: 'Shared', glyph: '🎒', tag: null, ghost: false, ...progress(part.shared) },
    { id: 'you', label: 'Your list', glyph: '🧳', tag: 'Private', ghost: false, ...progress(part.mine) },
    ...part.ghostLists.map((g) => ({
      id: g.member.id, label: memberName(g.member), glyph: initials(memberName(g.member)),
      tag: 'Ghost', ghost: true, ...progress(g.items),
    })),
  ]

  // Resolve the active list from `selected`.
  const active = resolveActive(selected, part, me, isWriter)

  function onToggle(item) { toggle.mutate({ id: item.id, packed: !item.packed }) }
  function onAdd() { setEditor({ scope: active.scope, ownerMemberId: active.ownerMemberId, item: null }) }
  function onEdit(item) { setEditor({ scope: item.scope, ownerMemberId: item.owner_member_id, item }) }
  function onRemove(item) { if (window.confirm('Remove this item?')) del.mutate(item.id) }

  return (
    <section className={styles.wrap}>
      <div className={styles.shell}>
        <PackingRail overall={progress(trip.packing_items)} entries={entries} selected={selected} onSelect={setSelected} />
        <PackingList
          title={active.title} glyph={active.glyph} scope={active.scope}
          items={active.items} members={trip.trip_members}
          canAdd={active.canManage} canManage={active.canManage} hint={active.hint}
          onAdd={onAdd} onToggle={onToggle} onEdit={onEdit} onRemove={onRemove} />
      </div>

      {editor && (
        <PackingItemEditorModal
          tripId={trip.id} memberId={me?.id ?? null}
          scope={editor.scope} ownerMemberId={editor.ownerMemberId}
          members={trip.trip_members} item={editor.item}
          onClose={() => setEditor(null)} />
      )}
    </section>
  )
}

// Which list is shown + whether the caller may manage it.
function resolveActive(selected, part, me, isWriter) {
  if (selected === 'you') {
    return { scope: 'personal', ownerMemberId: me?.id ?? null, items: part.mine,
      title: 'Your list', glyph: '🧳', hint: HINTS.personal, canManage: !!me }
  }
  const ghost = part.ghostLists.find((g) => g.member.id === selected)
  if (ghost) {
    return { scope: 'personal', ownerMemberId: ghost.member.id, items: ghost.items,
      title: `${memberName(ghost.member)}'s list`, glyph: '👤', hint: HINTS.ghost, canManage: isWriter }
  }
  return { scope: 'shared', ownerMemberId: null, items: part.shared,
    title: 'Shared', glyph: '🎒', hint: HINTS.shared, canManage: isWriter }
}
```
